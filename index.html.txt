    // Acoustic Sound Tuning using Tone.js
    function initSound() {
      try {
        if (!synth && window.Tone) {
          synth = new Tone.PolySynth(Tone.Synth, {
            oscillator: { type: 'sine' },
            envelope: { attack: 0.04, decay: 1.4, sustain: 0.08, release: 1.6 }
          }).toDestination();
          synth.volume.value = -12;
        }
      } catch (e) {
        console.warn("Audio Context init bypassed");
      }
    }

    function playChime(note = "E5") {
      if (isSoundMuted) return;
      try {
        if (window.Tone && Tone.context.state !== 'running') {
          Tone.start();
        }
        initSound();
        if (synth) synth.triggerAttackRelease(note, "8n");
      } catch (e) {}
    }

    function toggleMuteSound() {

      } else {
        icon.className = 'ph ph-speaker-high';
        playChime("A5");
        showToast('音效已啟動');
      }
    }

      };

      const orderTotal = cart.reduce((acc, it) => acc + (it.price * it.quantity), 0);
      const generatedOrderId = 'CS-' + new Date().getFullYear() + String(Math.floor(100000 + Math.random() * 900000));

      document.getElementById('receiptOrderId').innerText = generatedOrderId;
      document.getElementById('receiptCustName').innerText = name;
      document.getElementById('receiptCustPhone').innerText = phone;
      document.getElementById('receiptDeliveryDetail').innerText = deliveryDetail;
      document.getElementById('receiptPaymentMethod').innerText = paymentTitles[currentPaymentMethod] || '線上付款';
      document.getElementById('successOrderTotal').innerText = `NT$ ${orderTotal.toLocaleString()}`;

      closeCheckoutFormModal();
      document.getElementById('checkoutSuccessModal').classList.remove('hidden');

      cart = [];
      updateCartUI();
      playChime("B6");
      showToast('訂單已送出，感謝您的結緣');
    }

    function handleImageFileSelect(event, type, targetKey) {
      const file = event.target.files?.[0];
      if (!file) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        const newUrl = e.target.result;

        if (type === 'element') {
          if (ELEMENT_CONFIG[targetKey]) {
            ELEMENT_CONFIG[targetKey].macroImg = newUrl;
            const thumb = document.getElementById(`preview-thumb-el-${targetKey}`);
            if (thumb) thumb.src = newUrl;
            updateVisualizer();
            // Also update compendium card image if visible
            const compendiumImgs = document.querySelectorAll('#compendium img');
            const keyIndex = { metal: 0, wood: 1, water: 2, fire: 3, earth: 4 };
            if (compendiumImgs[keyIndex[targetKey]]) {
              compendiumImgs[keyIndex[targetKey]].src = newUrl;
            }
          }
        } else if (type === 'product') {
          const product = CURATED_PRODUCTS.find(p => p.id === targetKey);
          if (product) {
            product.image = newUrl;
            const thumb = document.getElementById(`preview-thumb-prod-${targetKey}`);
            if (thumb) thumb.src = newUrl;
            renderCuratedProducts();
          }
        }

        showToast('圖片已即時更新成功！');
        playChime("E6");
      };
      reader.readAsDataURL(file);
    }
